## Contract (implement exactly)
```rust
// src/i18n.rs
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Lang { Es, En, Pt }
impl Lang {
    pub const ALL: [Lang; 3] = [Lang::Es, Lang::En, Lang::Pt];
    pub fn name(self) -> &'static str;   // "Español" | "English" | "Português"
    pub fn from_locale(s: &str) -> Lang; // lowercase prefix "en"->En, "pt"->Pt, else Es
    pub fn from_env() -> Lang;           // first non-empty of LC_ALL, LC_MESSAGES, LANG; none -> Es
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Key { ... }
impl Key { pub const ALL: &'static [Key] = &[...every variant...]; }
pub fn t(lang: Lang, key: Key) -> &'static str {
    let [es, en, pt] = match key { Key::X => ["..", "..", ".."], ... };  // compiler enforces 3 per key
    match lang { Lang::Es => es, Lang::En => en, Lang::Pt => pt }
}
// src/model.rs
pub enum ErrorKind { Locked, ReadFailed(String), WatchTimeout, Backend(String) }  // derive Debug, Clone, PartialEq, Eq
// AppState::Error(String) -> AppState::Error(ErrorKind)
// ui
pub fn show(ui: &mut egui::Ui, state: &AppState, log: &[String], lang: Lang) -> Option<Command>;
```
Placeholders in templates: `{v}` (iOS version), `{name}`, `{model}`, `{detail}`; caller fills with `str::replace`.
